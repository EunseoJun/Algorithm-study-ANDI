from collections import deque

def solution(coin, cards):
    answer = 1
    n = len(cards)

    firstdeck = set(cards[:n//3])
    remain = deque(cards[n//3:])
    candidate = set()

    while len(remain) >= 2:
        two_cards = [remain.popleft(), remain.popleft()]
        candidate.update(two_cards)

        found = False

        # 1. 기존 카드끼리 (0코인)
        for card in firstdeck:
            target = n + 1 - card

            if target in firstdeck:
                firstdeck.remove(card)
                firstdeck.remove(target)
                found = True
                break

        if found:
            answer += 1
            continue

        # 2. 기존 카드 + 후보 카드 (1코인)
        if coin >= 1:
            for card in firstdeck:
                target = n + 1 - card

                if target in candidate:
                    firstdeck.remove(card)
                    candidate.remove(target)
                    coin -= 1
                    found = True
                    break

        if found:
            answer += 1
            continue

        # 3. 후보 카드끼리 (2코인)
        if coin >= 2:
            for card in candidate:
                target = n + 1 - card

                if target in candidate:
                    candidate.remove(card)
                    candidate.remove(target)
                    coin -= 2
                    found = True
                    break

        if found:
            answer += 1
            continue

        # 어떤 조합도 만들 수 없으면 종료
        break

    return answer
